    function buscarDadosPainel(idFieldset) {

        var arrIdFieldset = document.getElementById('hdnIdFieldset').value.split('|');

        if (!infraInArray(idFieldset, arrIdFieldset)) {
            var objAjaxBuscarDadosPainel = new ajaxBuscarDadosPainel();
            objAjaxBuscarDadosPainel.idFieldset = idFieldset;
            objAjaxBuscarDadosPainel.executar();

            if (document.getElementById('hdnIdFieldset').value != "") {
                document.getElementById('hdnIdFieldset').value += '|' + idFieldset;
            } else {
                document.getElementById('hdnIdFieldset').value = idFieldset;
            }
        }
    }

    function setCookie(nome, valor, validade) {
        var dataExp = new Date();
        dataExp.setDate(dataExp.getDate() + validade);
        var valorCookie = escape(valor) + ((validade == null) ? "" : "; expires=" + dataExp.toUTCString());
